// ⚙️ 선생님의 깃허브 정보 const OWNER = 'yjh6028-ship-it'; const REPO = 'class5-exam-statu'; const apiUrl = `https://api.github.com/repos/${OWNER}/${REPO}/contents/`; window.addEventListener('DOMContentLoaded', () => { const savedKeyword = localStorage.getItem('savedMemberId'); if (savedKeyword) { document.getElementById('searchKeyword').value = savedKeyword; } }); function handleEnter(e) { if(e.key === 'Enter') searchAndRedirect(); } function searchAndRedirect() { const keyword = document.getElementById('searchKeyword').value.trim(); const statusEl = document.getElementById('status'); const loaderEl = document.getElementById('loader'); if(!keyword) { statusEl.innerText = "수험번호를 입력해 주세요!"; statusEl.style.color = "#e74c3c"; return; } localStorage.setItem('savedMemberId', keyword); statusEl.innerText = "서버에서 최신 파일을 찾는 중입니다..."; statusEl.style.color = "#7f8c8d"; loaderEl.style.display = "block"; fetch(apiUrl) .then(response => { if (!response.ok) throw new Error('네트워크 응답에 문제가 있습니다.'); return response.json(); }) .then(data => { // 앱스 스크립트 양식에 맞게 검색 패턴 지정 (예: (105완료)) const searchPattern = `(${keyword}완료)`; // 1. PDF 파일이면서 파일명에 수험번호 패턴이 포함된 파일들만 필터링 const matchingFiles = data.filter(file => file.name.toLowerCase().endsWith('.pdf') && file.name.includes(searchPattern) ); if (matchingFiles.length > 0) { // 2. 날짜가 맨 앞에 있으므로 localeCompare 정렬 시 정확하게 시간순 정렬됩니다. matchingFiles.sort((a, b) => a.name.localeCompare(b.name)); // 맨 마지막 요소가 가장 최근에 생성된 파일 const latestPdf = matchingFiles[matchingFiles.length - 1]; statusEl.innerText = "파일 확인 완료! 화면을 이동합니다."; statusEl.style.color = "#27ae60"; // 💡 [핵심 수정] 상대 경로(./) 대신 깃허브가 제공하는 실제 파일 다운로드 URL로 이동합니다. // 만약 깃허브 페이지 내에서 상대적 경로 처리가 확실하다면 기존 코드를 쓰셔도 되지만, // 아래의 download_url을 쓰는 것이 가장 안전하고 확실합니다. window.location.href = latestPdf.download_url; } else { loaderEl.style.display = "none"; statusEl.innerText = `[${keyword}] 수험번호로 등록된 최신 자료가 없습니다.`; statusEl.style.color = "#e74c3c"; } }) .catch(error => { console.error('Error:', error); loaderEl.style.display = "none"; statusEl.innerText = "오류가 발생했습니다. 잠시 후 다시 시도해 주세요."; statusEl.style.color = "#e74c3c"; }); }
